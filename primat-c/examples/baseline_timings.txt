# captured once on this machine via:
#
#   python3 runfiles/benchmark.py

#
#   python3 -c "
#   import time, sys; sys.path.insert(0, '.')
#   from primat.backend import run_bbn
#   for net, kwargs in [('small', {'network':'small'}),
#                       ('large_amax8', {'network':'large','amax':8})]:

#       run_bbn({**kwargs, 'verbose': False, 'debug': False},
#               force_backend='python')

# weak-rate/electron-thermo caches warm (a cold first run pays a one-time

# matching `make bench`'s own `./primat-c` calls).
#
# python_small_s            = 0.386
# python_large_amax8_s      = 0.562
#
